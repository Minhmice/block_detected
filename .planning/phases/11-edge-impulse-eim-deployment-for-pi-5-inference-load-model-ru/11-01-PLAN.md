    Create `backend/app/services/eim_model.py`:

    ```python
    @dataclass(frozen=True)
    class EimModelStatus:
        path: Path
        exists: bool
        executable: bool
        error: Optional[str] = None

    def resolve_eim_path() -> Path:
        raw = os.getenv("EI_MODEL_PATH", "backend/models/block_detector.eim")
        path = Path(raw)
        if not path.is_absolute():
            path = _REPO_ROOT / path
        return path

    def validate_eim_model() -> EimModelStatus:
        path = resolve_eim_path()
        exists = path.is_file()
        executable = exists and os.access(path, os.X_OK)
        error = None
        if not exists:
            error = f"EIM model not found: {path}"
        elif not executable:
            error = f"EIM model not executable — run: chmod +x {path.relative_to(_REPO_ROOT) if path.is_relative_to(_REPO_ROOT) else path}"
        return EimModelStatus(path=path, exists=exists, executable=executable, error=error)

    def is_vision_mock_mode() -> bool:
        val = os.getenv("VISION_MOCK_MODE", "true").strip().lower()
        return val in ("1", "true", "yes", "on")
    ```

    Use `_REPO_ROOT = Path(__file__).resolve().parents[3]` (same as detection_loop).
